## 💻 2. ELIXIR CORE VIEW ENGINE IMPLEMENTATION

```elixir
defmodule CodeMantra.StreamAnimator do
  use GenServer

  # State Tracks: Sprite location, active monster state, and animation frame index
  def start_link(opts), do: GenServer.start_link(__MODULE__, %{frame: 0}, opts)
  def init(state), do: {:ok, state}

  @doc """
  Listens to state changes pushed down from the Progression Engine.
  Translates statistics into physical spatial positions along the World Tree.
  """
  def handle_cast({:queue_snapshot, user_id, player_stats, metrics}, state) do
    # 1. Map current file directory nesting depth to World Tree Height Coordinate (Y-Axis)
    file_path = metrics.raw_telemetry["buffer_context"]["file_path"] || "root"
    tree_y = calculate_tree_altitude(file_path)
    
    # 2. Translate current typing cadence (WPM) to Sprite Movement Speed (X-Axis)
    wpm = metrics.raw_telemetry["buffer_context"]["wpm_cadence"] || 0.0
    action_state = determine_sprite_action(wpm, metrics.raw_telemetry)

    # 3. Assemble the frame packet
    render_frame = %{
      frame_id: state.frame + 1,
      player: %{
        name: player_stats.title_str || "The Unknown Developer",
        level: player_stats.level,
        gold: player_stats.gold,
        sprite_x: rem(round(wpm * 5), 100), # Loop horizontally along the branch lane
        sprite_y: tree_y,
        action: action_state
      },
      environment: %{
        zone: determine_zone_tier(tree_y),
        active_bug_detected: !metrics.raw_telemetry["buffer_context"]["is_compilable"]
      }
    }

    # 4. Broadcast the structural canvas frame down to public web overlay viewers
    CodeMantraWeb.Endpoint.broadcast!("stream_view:#{user_id}", "canvas_update", render_frame)

    {:noreply, %{state | frame: state.frame + 1}}
  end

  # --- Internal Spatial Mapping Utilities ---
  defp calculate_tree_altitude(path) do
    # Count directory nesting slashes to calculate vertical climb displacement
    slashes = String.graphemes(path) |> Enum.count(&(&1 == "/"))
    min(slashes * 15, 100) # Cap at maximum tree altitude apex
  end

  defp determine_sprite_action(wpm, metrics) do
    cond do
      metrics["buffer_context"]["is_compilable"] == false -> "DEFENDING_COMPILER_ATTACK"
      wpm > 50.0 -> "BLAZING_FLOW_RUN"
      wpm > 15.0 -> "STEADY_GRIND_MARCH"
      true -> "IDLE_AWAITING_INPUT"
    end
  end

  defp determine_zone_tier(y) when y >= 75, do: "SKY_FORTRESS"
  defp determine_zone_tier(y) when y >= 50, do: "THE_BRANCHES"
  defp determine_zone_tier(y) when y >= 25, do: "THE_TRUNK"
  defp determine_zone_tier(y), do: "THE_ROOT_SYSTEM"
end
```
